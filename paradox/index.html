Basic Usage

examples/basic-usage/index.ts

Create a framework-neutral API runtime and bind one operation to a Next.js App Router route handler.

const runtime = createApiRuntime({
  definition: {
    id: "health-api",
    origin: "internal",
    protocol: "rest",
    basePath: "/api",
    endpoints: {
      health: {
        id: "health",
        kind: "http",
        operations: {
          "health.read": {
            id: "health.read",
            protocol: "http",
            intent: "read",
            method: "GET",
            path: "/health",
          },
        },
      },
    },
  },
  handlers: {
    "health.read": () => ({ body: { ok: true } }),
  },
});
